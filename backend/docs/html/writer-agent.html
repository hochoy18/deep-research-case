async def _outline(state, config):
    """Generate a structured report outline from the research topic and plan."""
    reasoning_model = state.get("reasoning_model") or configurable.answer_model

    agent = Agent(model_id=reasoning_model)
    agent.set_step_prompt(outline_instructions)

四、_draft：写草稿（:74-116）

同一个节点承担两种角色：首版草稿 和 修订草稿，通过 critic_feedback 区分。

核心代码

async def _draft(state, config):
    """Draft (or revise) the full report following the outline."""
    reasoning_model = state.get("reasoning_model") or configurable.answer_model
    feedback = state.get("critic_feedback", "")          # ★ 上轮 Critic 反馈
    outline_text = state.get("report_outline", "")
    is_revision = bool(feedback)                          # ★ 有 feedback 即为修订
    revision = state.get("revision_count", 0) + (1 if is_revision else 0)

    if is_revision:
        revision_context = (
            f"\n# 修订说明 (第 {revision} 次修订)\n"
            f"请根据以下审稿意见修改上一版草稿：\n\n"
            f"{feedback}\n\n"
            f"请逐条处理上述问题，优先修复 critical 和 major 级别的问题。"
            f"保留上版草稿中审稿人没有异议的内容。\n"
        )
        return_update = {"revision_count": revision, "critic_feedback": ""}
    else:
        revision_context = ""
        return_update = {}

    agent = Agent(model_id=reasoning_model)
    agent.set_step_prompt(draft_instructions)
    raw = await agent.astep(
        current_date=...,
        research_topic=...,
        research_proposal=...,
        outline=outline_text,
        summaries="\n---\n\n".join(state["web_search_result"]),
        revision_context=revision_context,                # 修订模式下注入审稿意见
    )
    draft = Post.extract_pattern(raw, pattern="markdown")

    return {**return_update, "report_draft": draft}

五、_critic_review：Critic 审稿（:119-178）

用 JsonAgent + CritiqueResult schema 输出结构化评分。

核心代码

async def _critic_review(state, config):
    """Critic reviews the draft and returns structured feedback."""
    reasoning_model = state.get("reasoning_model") or configurable.answer_model
    draft_text = state.get("report_draft", "")

    agent = JsonAgent(model_id=reasoning_model, keys=CritiqueResult)
    agent.set_step_prompt(critic_review_instructions)
    result = await agent.astep(
        research_topic=...,
        research_proposal=...,
        summaries=...,
        draft=draft_text,
    )

    # 降级：Critic LLM 失败 → 直接放行
    if not isinstance(result, CritiqueResult):
        return {
            "critic_feedback": "",
            "critic_score": 8.0,
            "ready_for_polish": True,
        }

    # 格式化 issues 为文本
    if result.issues:
        issues_text = "\n".join(
            f"- [{iss.severity.upper()}] {iss.location}: {iss.problem}\n"
            f"  建议: {iss.suggestion}"
            for iss in result.issues
        )
    else:
        issues_text = "无明显问题。"

    feedback = (
        f"## 审稿评分: {result.overall_rating}/10\n"
        f"## 综合评价: {result.summary}\n\n"
        f"## 具体问题:\n{issues_text}"
    )

    return {
        "critic_feedback": feedback,
        "critic_score": result.overall_rating,
        "ready_for_polish": result.ready_for_polish,
    }

六、_route_after_critic：四重退出保险（:181-233）

这是同步函数（无 LLM 调用），仅根据 state 决定下一步节点。

核心代码

def _route_after_critic(state, config) -> str:
    """决定：继续修改或进入终审润色。

    三重退出保险（设计理由：不能完全信任 LLM Critic 的自评，
    Critic 和 Writer 可能共用同一模型，存在"自己审自己"的偏差）：
      1. Critic 明确标记 ready_for_polish — LLM 判断质量合格
      2. critic_score >= 8.0 — 质量达标，即使 Critic 标记未 ready 也提前退出
      3. critic_score >= 6.0 且 revision_count >= 1 — 至少改过一次且评分及格
      4. revision_count >= max_revisions — 安全兜底，强制退出
    """
    revision = state.get("revision_count", 0)
    max_rev = state.get("max_revisions", DEFAULT_MAX_REVISIONS)
    ready = state.get("ready_for_polish", False)
    score = state.get("critic_score", 0.0)

    if ready:                return _CITE_AND_POLISH   # 条件 1
    if score >= 8.0:         return _CITE_AND_POLISH   # 条件 2
    if score >= 6.0 and revision >= 1: return _CITE_AND_POLISH  # 条件 3
    if revision >= max_rev:  return _CITE_AND_POLISH   # 条件 4
    return _DRAFT                                       # 否则回 draft 修订

七、_cite_and_polish：终稿润色（:236-283）

这是唯一流式的 WriterAgent 节点。两个步骤：LLM 润色 + 短 URL 替换。

核心代码

async def _cite_and_polish(state, config):
    """Finalise: LLM polish + replace short URLs with real URLs + deduplicate sources."""
    reasoning_model = state.get("reasoning_model") or configurable.answer_model
    draft_text = state.get("report_draft", "")

    # Step A — LLM polish pass (流式)
    agent = Agent(model_id=reasoning_model)
    agent.set_step_prompt(polish_instructions)

    emit_token = config.get("configurable", {}).get("_emit_token")
    if emit_token:
        async def on_token(text: str) -> None:
            await emit_token(text, "cite_and_polish")
        raw = await agent.astream_step(on_token, ...)
    else:
        raw = await agent.astep(...)

    polished = Post.extract_pattern(raw, pattern="markdown")

    # Step B — 短 URL → 真实 URL + 来源去重
    unique_sources = []
    for source in state.get("sources_gathered", []):
        if source["short_url"] in polished:
            polished = polished.replace(source["short_url"], source["value"])
            unique_sources.append(source)

    return {
        "messages": [AIMessage(content=polished)],     # 最终报告
        "sources_gathered": unique_sources,             # 去重后的引用源
    }

替换算法复杂度

代码：writer_agent.py:269-273

for source in state.get("sources_gathered", []):
    if source["short_url"] in polished:                    # O(N×M) substring check
        polished = polished.replace(source["short_url"], source["value"])
        unique_sources.append(source)

时间复杂度：O(N×M)，其中 N = 来源数，M = polished 长度。

潜在性能问题：str.replace() 会创建新字符串（不可变），对大文档 + 多源场景可能慢。 可考虑一次性正则替换：re.sub(pattern, lambda m: url_map[m.group(0)], polished)

14.1 模型复用风险

风险：Writer 和 Critic 共用同一模型 → LLM 倾向给自己打高分 / 不愿指出自己的问题。
缓解：四重退出保险 + 硬阈值 8.0 + 修订上限 3。
改进建议：用不同模型（如 critic 用更强的 reasoning 模型）。

14.4 潜在改进点

  1. Critic 与 Writer 共用模型：可能"自己审自己"。
    → 建议用不同 reasoning_model 给 Critic。
  2. URL 替换是 O(N×M)：大文档 + 多源场景可能慢。
    → 建议一次性 re.sub() 替换。
  3. 没有 token 成本控制：辩论循环最多 4×2 次 LLM 调用。
    → 建议根据任务长度动态调整 max_revisions。
  4. ready_for_polish=True 即使 score=0 也通过：
    → 改进：可加 score 硬下限（如 ready=True 要求 score >= 5）。
  5. 修订内容累积：每次 draft 都重写完整报告，而非增量修改。
    → 改进：可考虑保留上版 + 只修改 critic 指出问题的章节。
